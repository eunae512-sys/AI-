그리고 `getPreset` 함수 뒤(파일 끝)에 아래 순수함수를 통째로 추가한다. **알고리즘 확정 — 실행자는 발명하지 말고 이 코드를 그대로 사용:**

```typescript
// ─────────────────────────────────────────────────────────────────────────────
// auto 시퀀스 — 업종 편향 + (brand,topic,kind) 시드로 중간 슬라이드만 결정론 변주.
//   앵커 고정: [0]=masthead(표지) · [5]=type-hero(증거) · [6]=masthead(CTA)
//   중간 [1..4] 만 MID 팔레트에서 변주. Math.random 금지 — seed 산술만 (브라우저/노드 안전).
// ─────────────────────────────────────────────────────────────────────────────

/** 중간 슬롯에 쓰는 팔레트 (CLAUDE.md: 기존 5종 외 추가 금지 — masthead/type-hero 는 앵커 전용). */
const MID_PALETTE: SlideComposition[] = ["pillar-left", "paper-split", "overlay-card"];

/** 업종별 우세(편향) 컴포지션 — 중간 4슬롯 중 정확히 2회 등장. */
const INDUSTRY_BIAS: Record<Industry, SlideComposition> = {
  restaurant: "paper-split",
  dessert: "paper-split",
  stay: "paper-split",
  cafe: "overlay-card",
  local: "overlay-card",
  beauty: "pillar-left",
};

/** 작은 결정론 해시 — kind 문자열을 정수로 (seed 와 섞어 토픽/종류별 변주). */
function strHash(s: string): number {
  let h = 2166136261;
  for (let i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i);
    h = (h * 16777619) >>> 0;
  }
  return h;
}

/**
 * 7장 컴포지션 시퀀스를 반환한다.
 * 앵커 [0]/[5]/[6] 고정, 중간 [1..4] 만 업종 편향 + 시드로 변주.
 * 보장: bias 정확히 2슬롯 + 비-bias fill 2슬롯 → 항상 최소 2종, 같은 값 3연속 불가능.
 * 결정론: 같은 (industry, kind, seed) → 같은 출력.
 */
export function buildAutoSequence(
  industry: Industry,
  kind: string,
  seed: number,
): SlideComposition[] {
  const bias = INDUSTRY_BIAS[industry] ?? "paper-split";
  // seed 와 kind 를 섞은 변주 시드 — 같은 브랜드라도 토픽/종류마다 다른 배열.
  const s = (seed ^ strHash(kind)) >>> 0;

  // bias 를 박을 위치쌍 — 모두 비인접쌍이라 bias 2개가 절대 연속되지 않는다.
  //   → 중간 4슬롯에 같은 값 3연속이 구조적으로 불가능.
  const biasPairs: [number, number][] = [
    [0, 2],
    [1, 3],
    [0, 3],
  ];
  const [bA, bB] = biasPairs[s % biasPairs.length];

  // bias 를 뺀 나머지 2종 — 비-bias 슬롯 2개를 채운다 (서로 다른 값으로 → 최소 2종 보장 강화).
  const rest = MID_PALETTE.filter((c) => c !== bias); // 항상 길이 2
  const nonBias: number[] = [0, 1, 2, 3].filter((i) => i !== bA && i !== bB); // 항상 길이 2

  // 두 비-bias 슬롯에 rest[0]/rest[1] 을 시드로 순서만 바꿔 배정.
  const flip = (s >>> 3) % 2; // 0 또는 1
  const fill0 = rest[flip];
  const fill1 = rest[1 - flip];

  const mid: SlideComposition[] = [bias, bias, bias, bias];
  mid[nonBias[0]] = fill0;
  mid[nonBias[1]] = fill1;

  return ["masthead", mid[0], mid[1], mid[2], mid[3], "type-hero", "masthead"];
}
```

**불변식(실행자 — 코드 그대로 적용, 발명 금지):** `mid` 는 `[bias,bias,bias,bias]` 로 초기화 후 두 비-bias 슬롯(`nonBias[0..1]`)만 `fill0`/`fill1` 로 덮어쓴다. 결과는 **항상** bias 정확히 2개 + rest 2종 1개씩 = 최소 2종. biasPairs 가 전부 비인접쌍 `(0,2)/(1,3)/(0,3)` 이라 두 bias 가 인접하지 않으므로 **같은 값 3연속이 구조적으로 불가능**(별도 교정 루프 불필요). `mid` 는 처음부터 4개가 다 채워져 undefined·non-null 단언 없이 tsc 통과. `Industry`/`SlideComposition` 타입은 import 됨.

## 2. jiti 프로브 — buildAutoSequence 결정론·편향·앵커·제약 단언
임시 스크립트 `briq-app/scripts/probe-autovary.mjs` 생성 후 실행, 통과 확인 후 삭제(또는 .planning 에 남기지 않음):

```javascript
import path from "node:path";
import { fileURLToPath } from "node:url";
import { createJiti } from "jiti";

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);
const root = path.resolve(__dirname, "..");
const jiti = createJiti(__filename, { alias: { "@": root } });

const { buildAutoSequence } = await jiti.import("@/lib/brand/layout-preset.ts");
const { generateCardnewsCampaign } = await jiti.import("@/lib/cardnews/hook-generator.ts");

const INDUSTRIES = ["restaurant", "dessert", "stay", "cafe", "local", "beauty"];
const KINDS = ["신메뉴", "단골", "시즌", "리뷰"];
const ANCHOR = ["masthead", "type-hero", "masthead"]; // [0],[5],[6]
const TONE_COMPS = new Set(["paper-split", "overlay-card", "type-hero"]);

let fail = 0;
const assert = (cond, msg) => { if (!cond) { console.log("FAIL:", msg); fail++; } };

// 시드는 같은 브랜드 가정: 고정 base 에 kind 만 섞이는지 보려 임의 seed 하나 고정.
const SEED = 123456789;

for (const ind of INDUSTRIES) {
  const biasCount = {};
  for (const kind of KINDS) {
    const seq = buildAutoSequence(ind, kind, SEED);
    // (d) 앵커 고정
    assert(seq[0] === "masthead" && seq[5] === "type-hero" && seq[6] === "masthead",
      `anchor ${ind}/${kind} -> ${seq.join(",")}`);
    // (d) 길이 7
    assert(seq.length === 7, `len ${ind}/${kind}`);
    // (d) 3연속 금지 (전체)
    for (let i = 2; i < seq.length; i++)
      assert(!(seq[i] === seq[i-1] && seq[i-1] === seq[i-2]), `3연속 ${ind}/${kind} @${i}`);
    // (d) 중간 최소 2종
    const mid = seq.slice(1, 5);
    assert(new Set(mid).size >= 2, `mid<2종 ${ind}/${kind} -> ${mid.join(",")}`);
    // (c) 결정론
    const seq2 = buildAutoSequence(ind, kind, SEED);
    assert(JSON.stringify(seq) === JSON.stringify(seq2), `결정론 ${ind}/${kind}`);
    // 편향 집계
    const biasMap = { restaurant:"paper-split", dessert:"paper-split", stay:"paper-split",
                      cafe:"overlay-card", local:"overlay-card", beauty:"pillar-left" };
    mid.forEach((c) => { biasCount[c] = (biasCount[c]||0)+1; });
    assert(mid.filter((c) => c === biasMap[ind]).length >= 2,
      `편향 ${ind}/${kind}: ${biasMap[ind]} <2 in ${mid.join(",")}`);
    console.log(`${ind.padEnd(11)} ${kind.padEnd(4)} -> ${seq.join(", ")}`);
  }
  // (a) 같은 브랜드 다른 토픽 = 중간 배열 다름 (적어도 KINDS 중 2개는 서로 다름)
  const mids = KINDS.map((k) => buildAutoSequence(ind, k, SEED).slice(1,5).join(","));
  assert(new Set(mids).size >= 2, `토픽변주 ${ind}: ${[...new Set(mids)].join(" | ")}`);
}

// (b) 생성기 통합 — composition 변주 + paperTone 이 톤-쓰는 컴포지션에만
const BRAND = {
  id: "probe-brand", name: "프로브", handle: "probe", wordmark: "PROBE",
  industry: "restaurant", industryLabel: "음식점", city: "서울", mood: "warm",
  signature: "코스", saveRate: 4.0,
}; // 실제 Brand 필드 — types/index.ts 참고해 필수 필드 채울 것(빠진 필드 tsc 와 무관, 런타임만).
for (const topic of ["봄나물 신메뉴", "단골 감사 이벤트"]) {
  const camp = generateCardnewsCampaign(topic, BRAND);
  const comps = camp.slides.map((s) => s.composition);
  console.log(`gen "${topic}" comps:`, comps.join(", "));
  camp.slides.forEach((s, i) => {
    if (s.paperTone) assert(TONE_COMPS.has(s.composition),
      `톤배선 위반 slide${i+1} comp=${s.composition} tone=${s.paperTone}`);
  });
}

console.log(fail === 0 ? "\n✅ ALL PASS" : `\n❌ ${fail} FAIL`);
process.exit(fail === 0 ? 0 : 1);
```
